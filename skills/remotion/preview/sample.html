import {Composition} from 'remotion';
import {MenuReel} from './MenuReel';

export const RemotionRoot = () => {
  return (
    <Composition
      id="MenuReel"

      durationInFrames={150}
      fps={30}
      width={1080}
      height={1920}
      defaultProps={{
        title: 'Cà phê muối',

src/MenuReel.tsx: mỗi cảnh một TransitionSeries.Sequence, cỡ chữ ≥ 84 px (tiêu đề) và ≥ 44 px (chữ phụ) cho khổ rộng 1080, lề an toàn 100/80 px
import {AbsoluteFill, interpolate, useCurrentFrame} from 'remotion';
import {TransitionSeries, linearTiming} from '@remotion/transitions';
import {fade} from '@remotion/transitions/fade';

type Props = {
  readonly title: string;
  readonly subtitle: string;
  readonly price: string;
};

const Scene = ({big, small, bg}: {big: string; small: string; bg: string}) => {
  const frame = useCurrentFrame();
  const rise = interpolate(frame, [0, 20], [60, 0], {extrapolateRight: 'clamp'});
  return (
    <AbsoluteFill style={{background: bg, padding: '100px 80px', justifyContent: 'center'}}>
      <div style={{fontSize: 120, fontWeight: 800, color: '#fff', transform: `translateY(${rise}px)`}}>{big}</div>
      <div style={{fontSize: 52, color: '#f3e6d0', marginTop: 32}}>{small}</div>
    </AbsoluteFill>
  );
};

// 3 cảnh x 60 khung - 2 chuyển cảnh x 15 khung = 150 khung (5 giây ở 30 fps)
export const MenuReel = ({title, subtitle, price}: Props) => (
  <TransitionSeries>
    <TransitionSeries.Sequence durationInFrames={60}>
      <Scene big="Món mới" small={subtitle} bg="#3b2a20" />
    </TransitionSeries.Sequence>
    <TransitionSeries.Transition presentation={fade()} timing={linearTiming({durationInFrames: 15})} />
    <TransitionSeries.Sequence durationInFrames={60}>
      <Scene big={title} small="Ngọt dịu, mặn nhẹ" bg="#1f6f6b" />
    </TransitionSeries.Sequence>
    <TransitionSeries.Transition presentation={fade()} timing={linearTiming({durationInFrames: 15})} />
    <TransitionSeries.Sequence durationInFrames={60}>
      <Scene big={price} small="Ghé quán hôm nay" bg="#8a3b2a" />
    </TransitionSeries.Sequence>
  </TransitionSeries>
);
