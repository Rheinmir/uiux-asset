src/slug.ts
import { DEFAULT_MAX_LENGTH, DEFAULT_SEPARATOR } from './constants'
import type { SlugOptions } from './types'

export function slugify(input: string, options: SlugOptions = {}): string {
  const { separator = DEFAULT_SEPARATOR, maxLength = DEFAULT_MAX_LENGTH } = options
  // NFD tách dấu khỏi chữ cái; đ/Đ không có dạng tách nên phải đổi tay
  return input
    .normalize('NFD')
    .replace(/[\u0300-\u036F]/g, '')
    .replace(/đ/gi, 'd')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, separator)
    .replace(new RegExp(`^${separator}|${separator}$`, 'g'), '')
    .slice(0, maxLength)
}
src/slug.test.ts
import { describe, expect, it } from 'vitest'
import { slugify } from './slug'

describe('slugify', () => {
  it('strips Vietnamese diacritics', () => {
    expect(slugify('Lập trình & tự động hóa')).toBe('lap-trinh-tu-dong-hoa')
  })

  it('handles đ and trims separators', () => {
    expect(slugify('  Đường dẫn  Đẹp!! ')).toBe('duong-dan-dep')
  })
})
